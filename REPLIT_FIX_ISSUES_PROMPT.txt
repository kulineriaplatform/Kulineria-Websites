STEP 1.4: Frontend - Setup CSRF Token Handling
File: /lib/api-utils.ts

Replace entire file content with:

------BEGIN CODE------
import axios from 'axios';

const API_URL = process.env.NEXT_PUBLIC_API_URL || 'http://localhost:8000/api';

// Create axios instance with credentials
export const apiClient = axios.create({
  baseURL: API_URL,
  withCredentials: true, // CRITICAL: Include cookies in all requests
  headers: {
    'Accept': 'application/json',
    'Content-Type': 'application/json',
  },
});

// Get CSRF token from backend
export const getCSRFToken = async () => {
  try {
    await apiClient.get('/sanctum/csrf-cookie', {
      baseURL: process.env.NEXT_PUBLIC_API_URL?.replace('/api', '') || 'http://localhost:8000',
    });
  } catch (error) {
    console.error('Failed to get CSRF token:', error);
  }
};

// Initialize CSRF token on app load
export const initializeAuth = async () => {
  await getCSRFToken();
};

// Request interceptor to ensure CSRF token before each request
apiClient.interceptors.request.use(
  async (config) => {
    // Get fresh CSRF token before each request
    await getCSRFToken();
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// Response interceptor for error handling
apiClient.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 419) {
      // Token expired, refresh and retry
      console.warn('CSRF token expired, refreshing...');
      window.location.reload();
    }
    return Promise.reject(error);
  }
);

export default apiClient;
------END CODE------

STEP 2.6: Create Middleware for Response Optimization
File: /middleware.ts (create new file at root)

------BEGIN CODE------
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';

export function middleware(request: NextRequest) {
  const response = NextResponse.next();
  
  // Add security headers
  response.headers.set('X-Content-Type-Options', 'nosniff');
  response.headers.set('X-Frame-Options', 'DENY');
  response.headers.set('X-XSS-Protection', '1; mode=block');
  
  // Add caching headers for static assets
  if (request.nextUrl.pathname.startsWith('/_next/static/')) {
    response.headers.set('Cache-Control', 'public, max-age=31536000, immutable');
  }
  
  return response;
}

export const config = {
  matcher: ['/((?!api|_next/static|_next/image|favicon.ico).*)'],
};
------END CODE------

STEP 2.7: Add Request Timeout & Retry Logic
File: /lib/api-utils.ts (UPDATE the interceptors section)

Add after response interceptor:

------BEGIN CODE------
// Retry logic for failed requests
let retryCount = 0;
const MAX_RETRIES = 3;

apiClient.interceptors.response.use(
  (response) => {
    retryCount = 0;
    return response;
  },
  async (error) => {
    const config = error.config;
    
    // Token expired
    if (error.response?.status === 419) {
      console.warn('CSRF token expired, refreshing...');
      window.location.reload();
      return Promise.reject(error);
    }
    
    // Retry on server errors (5xx) with exponential backoff
    if (error.response?.status >= 500 && retryCount < MAX_RETRIES) {
      retryCount++;
      const delay = Math.pow(2, retryCount) * 1000; // 2s, 4s, 8s
      
      await new Promise(resolve => setTimeout(resolve, delay));
      return apiClient(config);
    }
    
    retryCount = 0;
    return Promise.reject(error);
  }
);
------END CODE------

STEP 4.5: Common Errors & Fixes

ERROR: "CSRF token mismatch"
FIX: 
- Ensure withCredentials: true in api-utils.ts
- Ensure 'supports_credentials' => true in cors.php
- Check SESSION_SAME_SITE=Lax in .env

ERROR: "419 Token Mismatch"
FIX:
- Run: php artisan config:cache
- Clear browser cookies
- Refresh page and try again

ERROR: "Slow loading (>2s response)"
FIX:
- Check if config:cache was run
- Verify APP_DEBUG=false
- Check if route:cache was run
- Monitor memory: free -h

ERROR: "502 Bad Gateway"
FIX:
- Check memory: ps aux | grep php
- Restart: kill $BACKEND_PID
- Increase memory: -d memory_limit=512M
